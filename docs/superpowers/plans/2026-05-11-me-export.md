  /api/v1/me/export:
    get:
      operationId: exportMe
      summary: Download a JSON bundle of the user's data.
      description: |
        Returns every character, ship, and game session owned by the
        authenticated user as a single versioned JSON envelope. Intended
        for GDPR-style data portability per PRD §6.

        The response is served with `Content-Disposition: attachment` and a
        filename of the form `coriolis-export-<userId>-<YYYY-MM-DD>.json`,
        so a browser fetch triggers a file download rather than a render.

        The envelope's `$schemaVersion` is a string literal `"1"` — future
        shape changes ship as `"2"` etc., so a downstream re-import path
        can branch on it deterministically.

        Bundle scope:
        - **Characters** with nested `items`, `statusFlags`, `talents`.
        - **Ships** with nested `modules` and `crew`.
        - **Game sessions** with nested `npcs` (each with `weapons` / `armor`),
          `encounters`, and each encounter's `combatants`.

        Out of scope: rolls log, share-link tokens, push records, user
        settings. Those are tracked as deferred follow-ups in the design
        doc.
      tags: [users]
      security:
        - cookieAuth: []
        - bearerAuth: []
      responses:
        '200':
          description: The user's data bundle.
          headers:
            Content-Disposition:
              schema:
                type: string
              description: |
                Always `attachment; filename="coriolis-export-<userId>-<date>.json"`.
                Clients should parse the filename when triggering downloads.
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/MeExportBundle'
        '401':
          description: No active session
          content:
            application/problem+json:
              schema:
                $ref: '#/components/schemas/Problem'
  ```

  export const EXPORT_REPO = Symbol('EXPORT_REPO');

  /**
   * Minimal user envelope used inside the export bundle. Mirrors the
   * MeExportUser OpenAPI schema. Excludes email by design.
   */
  export interface MeExportUserDto {
    id: string;
    displayName: string | null;
    locale: string;
  }

  /**
   * The bundle shape returned by ExportMeHandler. The nested array types
   * use `unknown` because they are owned by the Characters / Ships /
   * GameSessions bounded contexts and rebuilt verbatim from Prisma. The
   * OpenAPI schema is the contract — runtime validation by
   * `express-openapi-validator` is the safety net.
   */
  export interface MeExportBundle {
    $schemaVersion: '1';
    exportedAt: string; // ISO-8601 UTC
    user: MeExportUserDto;
    characters: unknown[];
    ships: unknown[];
    gameSessions: unknown[];
  }

  export interface IExportRepo {
    /**
     * Read the user + their characters, ships, and game sessions in the
     * shape expected by the OpenAPI export schemas. Soft-deleted records
     * are excluded by `deletedAt: null` filters in the adapter. Each
     * include is scoped to its bounded context's owned children.
     */
    loadBundleFor(userId: string): Promise<MeExportBundle>;
  }
  ```

## Task 4 (Block 2): Backend adapter

**Files:** `apps/backend/src/modules/users/infra/export.repo.ts` (create).

- [ ] **Step 4.1:** Create the adapter:

  ```ts
  /**
   * WHY this file exists:
   * Infrastructure adapter — the Prisma-backed implementation of
   * IExportRepo. The only place in the users bounded context that
   * cross-reads characters / ships / gameSessions tables. Every model
   * uses `ownerId` as the user FK (verified against
   * prisma/schema.prisma at design time).
   *
   * Soft-deletion: users that hit DELETE /me have `deletedAt` stamped;
   * their data persists 30 days for share-link readers before a daily
   * cron hard-deletes. For an export, we filter `deletedAt: null` on
   * the user lookup but include their characters / ships / sessions
   * unconditionally — the caller is authenticated as that user (live
   * session), so the user row cannot be soft-deleted in this code
   * path.
   */
  import { Injectable } from '@nestjs/common';

  import { PrismaService } from '../../../common/prisma/prisma.service';

  import type { IExportRepo, MeExportBundle, MeExportUserDto } from '../domain/user.ports';

  @Injectable()
  export class ExportRepo implements IExportRepo {
    constructor(private readonly prisma: PrismaService) {}

    async loadBundleFor(userId: string): Promise<MeExportBundle> {
      const user = await this.prisma.user.findUniqueOrThrow({
        where: { id: userId },
        select: { id: true, displayName: true, locale: true },
      });

      const characters = await this.prisma.character.findMany({
        where: { ownerId: userId },
        include: {
          items: true,
          statusFlags: true,
          talents: true,
        },
      });

      const ships = await this.prisma.ship.findMany({
        where: { ownerId: userId },
        include: {
          modules: true,
          crew: true,
        },
      });

      const gameSessions = await this.prisma.gameSession.findMany({
        where: { ownerId: userId },
        include: {
          npcs: { include: { weapons: true, armor: true } },
          encounters: { include: { combatants: true } },
        },
      });

      const userDto: MeExportUserDto = {
        id: user.id,
        displayName: user.displayName,
        locale: user.locale,
      };

      return {
        $schemaVersion: '1',
        exportedAt: new Date().toISOString(),
        user: userDto,
        characters,
        ships,
        gameSessions,
      };
    }
  }
  ```

  > **Schema sanity step:** if `pnpm --filter @app/backend typecheck`

  > does not exist (e.g. `Character.talents` not being a relation),

  > include. The spec's `MeExportBundle` allows missing children
  > only if you defined a wrapper schema in Block 1; otherwise
  > leave the include in but stub the empty case at runtime if Prisma
  > confirms the relation exists with a different name.

  /**
   * WHY this file exists:
   * Read-side handler for the JSON export. Receives an ExportMeQuery,
   * delegates to IExportRepo, returns the bundle untouched. No mutation,
   * no business logic — the repo composes the cross-resource query and
   * shapes the envelope. The handler is the seam where CQRS gives us a
   * single test target.
   */
  import { Inject } from '@nestjs/common';
  import { IQueryHandler, QueryHandler } from '@nestjs/cqrs';

  import { EXPORT_REPO } from '../domain/user.ports';

  import { ExportMeQuery } from './export-me.query';

  import type { IExportRepo, MeExportBundle } from '../domain/user.ports';

  @QueryHandler(ExportMeQuery)
  export class ExportMeHandler implements IQueryHandler<ExportMeQuery, MeExportBundle> {
    constructor(@Inject(EXPORT_REPO) private readonly repo: IExportRepo) {}

    async execute(query: ExportMeQuery): Promise<MeExportBundle> {
      return this.repo.loadBundleFor(query.userId);
    }
  }
  ```

    /**
     * GET /api/v1/me/export
     *
     * Streams the user's full data bundle as a downloadable JSON
     * attachment. The bundle is buffered (built in-memory) for v1 —
     * see the design doc for the streaming follow-up trigger.
     */
    @Get('export')
    async exportMe(
      @Req() req: AuthenticatedRequest,
      @Res({ passthrough: true }) res: Response,
    ): Promise<MeExportBundle> {
      const bundle = await this.queryBus.execute<ExportMeQuery, MeExportBundle>(
        new ExportMeQuery(req.userId),
      );
      const today = new Date().toISOString().slice(0, 10); // YYYY-MM-DD
      const filename = `coriolis-export-${req.userId}-${today}.json`;
      res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);
      return bundle;
    }
  ```

  > `passthrough: true` lets Nest serialise the return value while we
  > still mutate the response headers — the standard pattern. Without
  > it, returning a value while injecting `@Res()` short-circuits Nest's
  > serializer.

## Task 8 (Block 3): Web — extend `useMeApi`

**Files:** `apps/web/app/composables/useMeApi.ts`.

- [ ] **Step 8.1:** Open `apps/web/app/composables/useMeApi.ts`. Add `exportMe` to the imports + interface + return object. Full file after the edit:

  ```ts
  /**
   * useMeApi — /api/v1/me HTTP layer for apps/web.
   *
   * All calls to the authenticated-user profile endpoint go through this
   * composable. Internally delegates to the generated `@app/api-client-ts`
   * SDK functions. Errors are thrown as the raw `Problem` object (RFC 9457)
   * so callers can inspect `problem.status`, `problem.detail`, etc.
   *
   * Re-exports `Me` and `MeUpdate` so pages import from one place.
   */

  import { getMe, updateMe, deleteMe, exportMe } from '@app/api-client-ts';
  import type { Me, MeUpdate, Problem } from '@app/api-client-ts';

  export interface MeApiClient {
    get(): Promise<Me>;
    update(body: MeUpdate): Promise<Me>;
    deleteMe(): Promise<void>;
    /**
     * Returns the response as a Blob plus the filename suggested by the
     * server (parsed from Content-Disposition). The download trigger is
     * the caller's responsibility.
     */
    exportMe(): Promise<{ blob: Blob; filename: string }>;
  }

  const DEFAULT_EXPORT_FILENAME = 'coriolis-export.json';

  function parseFilename(header: string | undefined): string {
    if (!header) return DEFAULT_EXPORT_FILENAME;
    const match = header.match(/filename="([^"]+)"/);
    return match?.[1] ?? DEFAULT_EXPORT_FILENAME;
  }

  export function useMeApi(): MeApiClient {
    return {
      async get() {
        const res = await getMe();
        // eslint-disable-next-line @typescript-eslint/only-throw-error
        if (res.error) throw res.error as Problem;
        return res.data;
      },

      async update(body) {
        const res = await updateMe({ body });
        // eslint-disable-next-line @typescript-eslint/only-throw-error
        if (res.error) throw res.error as Problem;
        return res.data;
      },

      async deleteMe() {
        const res = await deleteMe();
        // eslint-disable-next-line @typescript-eslint/only-throw-error
        if (res.error) throw res.error as Problem;
      },

      async exportMe() {
        // The generated SDK returns the parsed body by default. Override
        // parseAs to 'blob' so we can stream it straight to a download
        // anchor without round-tripping through JSON.stringify.
        const res = await exportMe({ parseAs: 'blob' });
        // eslint-disable-next-line @typescript-eslint/only-throw-error
        if (res.error) throw res.error as Problem;
        const blob = res.data as unknown as Blob;
        const filename = parseFilename(
          res.response?.headers.get('Content-Disposition') ?? undefined,
        );
        return { blob, filename };
      },
    };
  }

  export { type Me, type MeUpdate, type Problem } from '@app/api-client-ts';
  ```

  > **`parseAs` option:** the generated client wraps `@hey-api/client-*`
  > or `openapi-fetch` depending on the codegen config. Look at one of
  > the other call sites in `apps/web` that pulls a binary (if any
  > exist) for the exact option name. If `parseAs` does not work,
  > equivalent options are `responseType: 'blob'` (axios-style) or
  > `accept: '*/*'` + manual `res.response?.blob()`. Pick whichever the
  > existing SDK accepts and document the choice in the commit message.
